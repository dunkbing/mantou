import { Card, CardGrid } from '@astrojs/starlight/components';

```bash
bun add mantou-ui
```

```tsx
import { Text, View, createSignal, run } from "mantou-ui";

function Counter() {
  const [count, setCount] = createSignal(0);
  return (
    <View style={{ padding: 24, gap: 12, alignItems: "flex-start" }}>
      <Text style={{ fontSize: 32, fontWeight: 700 }}>{count()}</Text>
      <View style={{ padding: 12, borderRadius: 8, backgroundColor: "#2563eb" }} onPress={() => setCount(count() + 1)}>
        <Text style={{ color: "#fff", fontWeight: 600 }}>Add one</Text>
      </View>
    </View>
  );
}

run(() => <Counter />, { title: "Counter" });
```

## What you get

<CardGrid>
  <Card title="Fast by construction" icon="rocket">
    The window runs on its own thread. Scrolling, resizing and animations don't wait for your code, and an idle app uses no CPU.
  </Card>
  <Card title="Familiar" icon="pen">
    Solid signals and JSX, flexbox layout, and the npm packages you already use, on Bun.
  </Card>
  <Card title="The parts apps need" icon="puzzle">
    Text input with IME, virtualized lists, images and SVG, custom fonts, shadows, animations, menus, dialogs, clipboard, drag and drop.
  </Card>
  <Card title="Accessible" icon="approve-check">
    Screen readers see your UI through AccessKit on macOS, Windows and Linux.
  </Card>
  <Card title="Testable" icon="seti:checkbox">
    Test against the real layout and text engine with `bun test`, no window needed.
  </Card>
  <Card title="One command to ship" icon="download">
    `mantou build` makes one executable; `mantou package` makes a macOS app and disk image.
  </Card>
</CardGrid>
